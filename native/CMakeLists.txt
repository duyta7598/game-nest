cmake_minimum_required(VERSION 3.20)
project(gamenest_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
add_executable(gamenest-engine engine.cpp)
target_link_libraries(gamenest-engine PRIVATE LibtorrentRasterbar::torrent-rasterbar)
set_target_properties(gamenest-engine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
set_target_properties(gamenest-engine PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin")
install(TARGETS gamenest-engine RUNTIME DESTINATION bin)
